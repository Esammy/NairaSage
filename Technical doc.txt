1. System Overview
1.1 Architecture Philosophy
NairaSage is a full-stack, AI-native financial platform built on a decoupled architecture: a Next.js TypeScript frontend communicates with a Django REST API backend over HTTPS, while a separate AI orchestration layer (Django services + LangChain) handles multi-agent coordination. Background work (opportunity scouting, cron triggers, autonomous execution) runs on Django-RQ workers backed by Redis.

The system is designed to be LLM-agnostic at the configuration level. Switching between Mistral (default) and OpenAI requires only a change in the .env file — no code changes needed. LangChain provides the abstraction layer that makes this possible.

1.3 LLM Configuration (.env Switching)
All AI model selection is controlled by environment variables. The LangChain abstraction ensures the same agent code runs regardless of which provider is active:

# Set to: mistral | openai
LLM_PROVIDER=mistral


# ── Mistral Configuration ───────────────────────────────
MISTRAL_API_KEY=your_mistral_api_key_here
MISTRAL_MODEL=mistral-large-latest          # or mistral-medium, open-mistral-7b


# ── OpenAI Configuration (fallback / alternative) ───────
OPENAI_API_KEY=your_openai_api_key_here
OPENAI_MODEL=gpt-4o                        # or gpt-4-turbo, gpt-3.5-turbo


# ── Shared LLM Settings ──────────────────────────────────
LLM_TEMPERATURE=0.2                        # Low = more deterministic financial advice
LLM_MAX_TOKENS=4096
LLM_TIMEOUT=60


# core/llm_factory.py
import os
from langchain_mistralai import ChatMistralAI
from langchain_openai import ChatOpenAI
from langchain_core.language_models import BaseChatModel


def get_llm(temperature: float = None) -> BaseChatModel:
    provider = os.getenv('LLM_PROVIDER', 'mistral').lower()
    temp = temperature or float(os.getenv('LLM_TEMPERATURE', '0.2'))


    if provider == 'openai':
        return ChatOpenAI(
            model=os.getenv('OPENAI_MODEL', 'gpt-4o'),
            temperature=temp,
            max_tokens=int(os.getenv('LLM_MAX_TOKENS', '4096')),
            openai_api_key=os.getenv('OPENAI_API_KEY'),
        )
    # Default: Mistral
    return ChatMistralAI(
        model=os.getenv('MISTRAL_MODEL', 'mistral-large-latest'),
        temperature=temp,
        mistral_api_key=os.getenv('MISTRAL_API_KEY'),
    )

4. AI Agent Architecture
4.1 Base Agent Class
# agents/base_agent.py


from abc import ABC, abstractmethod
from typing import Any, Dict, Optional
from langchain_core.messages import HumanMessage, SystemMessage, AIMessage
from langchain.memory import ConversationSummaryBufferMemory
from core.llm_factory import get_llm
from apps.users.models import FinancialDNA


class BaseAgent(ABC):
    """Abstract base for all NairaSage specialist agents."""


    agent_name: str = 'base'
    system_prompt: str = ''


    def __init__(self, user_id: int):
        self.user_id  = user_id
        self.llm      = get_llm()
        self.dna      = FinancialDNA.objects.select_related('user').get(user_id=user_id)
        self.memory   = self._load_memory()


    def _load_memory(self) -> ConversationSummaryBufferMemory:
        """Load agent-specific conversation memory from Financial DNA store."""
        mem = ConversationSummaryBufferMemory(
            llm=self.llm,
            max_token_limit=2000,
            return_messages=True,
        )
        # Restore saved summary from DNA if it exists
        saved = self.dna.agent_memory.get(self.agent_name, {})
        if saved.get('summary'):
            mem.moving_summary_buffer = saved['summary']
        return mem


    def _save_memory(self):
        """Persist memory summary back to Financial DNA."""
        self.dna.agent_memory[self.agent_name] = {
            'summary': self.memory.moving_summary_buffer
        }
        self.dna.save(update_fields=['agent_memory', 'updated_at'])


    def _get_dna_context(self) -> str:
        """Serialize relevant DNA fields as context for the system prompt."""
        return (
            f'User profile: income={self.dna.monthly_income} NGN/month, '
            f'risk_tolerance={self.dna.risk_tolerance}, '
            f'goal={self.dna.primary_goal}, '
            f'literacy_score={self.dna.financial_literacy_score}/100.'
        )


    def build_messages(self, user_input: str) -> list:
        history = self.memory.load_memory_variables({}).get('history', [])
        return [
            SystemMessage(content=self.system_prompt + '\n\n' + self._get_dna_context()),
            *history,
            HumanMessage(content=user_input),
        ]


    @abstractmethod
    def run(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """Execute the agent. Return structured result dict."""
        pass


    def chat(self, user_input: str) -> str:
        """Standard chat interface with memory persistence."""
        messages = self.build_messages(user_input)
        response = self.llm.invoke(messages)
        self.memory.save_context({'input': user_input}, {'output': response.content})
        self._save_memory()
        return response.content

  "should I buy MTNN stock?"         -> ["investment_advisor", "risk_assessor"]
  "plan my retirement"               -> ["life_planner", "savings_optimizer", "risk_assessor"]
  "I keep impulse spending"          -> ["behavioral_coach", "accountant"]
  "what opportunities exist for me?" -> ["opportunity_scout"]
  "how much tax do I owe?"           -> ["tax_advisor", "accountant"]
'''


class CoordinatorAgent:
    def __init__(self, user_id: int):
        self.user_id = user_id
        self.llm     = get_llm(temperature=0.0)  # deterministic routing


    def route(self, user_message: str) -> List[str]:
        """Determine which agents to invoke for this message."""
        import json
        from langchain_core.messages import SystemMessage, HumanMessage
        response = self.llm.invoke([
            SystemMessage(content=ROUTING_PROMPT),
            HumanMessage(content=user_message),
        ])
        try:
            return json.loads(response.content)
        except Exception:
            return ['accountant']  # safe fallback


    def dispatch(self, user_message: str) -> dict:
        """Route to agents, run them, merge results."""
        agents_needed = self.route(user_message)
        results = {}
        for agent_name in agents_needed:
            AgentClass = AGENT_REGISTRY.get(agent_name)
            if AgentClass:
                agent = AgentClass(user_id=self.user_id)
                results[agent_name] = agent.run({'query': user_message})


        return self._synthesize(user_message, results)


    def _synthesize(self, query: str, agent_results: dict) -> dict:
        """Combine multi-agent results into unified response."""
        from langchain_core.messages import SystemMessage, HumanMessage
        context = '\n\n'.join([
            f'[{name}]: {result.get("summary","")}' 
            for name, result in agent_results.items()
        ])
        synthesis = self.llm.invoke([
            SystemMessage(content='Synthesize these specialist agent reports into a single, clear, actionable response for the user. Be concise and Nigerian-context aware.'),
            HumanMessage(content=f'User asked: {query}\n\nAgent reports:\n{context}'),
        ])
        return {
            'response': synthesis.content,
            'agents_used': list(agent_results.keys()),
            'raw': agent_results,
        }

# agents/accountant/agent.py


from agents.base_agent import BaseAgent
from apps.transactions.models import Transaction
from apps.accounts.models import BankAccount
from langchain.tools import tool
from langchain.agents import create_tool_calling_agent, AgentExecutor
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder
import pandas as pd
from datetime import date, timedelta


class AccountantAgent(BaseAgent):
    agent_name = 'accountant'
    system_prompt = '''
    You are a Nigerian personal accountant and financial analyst.
    You have direct access to the user's real transaction data.
    Always cite specific numbers from their actual data.
    Speak in clear, plain English. Use NGN for amounts.
    Flag unusual patterns, recurring charges, and waste.
    '''


    def run(self, input_data: dict) -> dict:
        query = input_data.get('query', '')


        # Fetch 90 days of user transactions
        txns = Transaction.objects.filter(
            account__user_id=self.user_id,
            date__gte=date.today() - timedelta(days=90)
        ).values('amount','type','category','description','date','merchant')


        df = pd.DataFrame(list(txns))
        summary = self._build_summary(df)


        prompt = f'''
        Transaction summary (last 90 days):\n{summary}\n\n
        User query: {query}
        '''
        response = self.chat(prompt)
        return {'summary': response, 'data': summary}


    def _build_summary(self, df: pd.DataFrame) -> str:
        if df.empty:
            return 'No transactions found.'
        credits = df[df['type']=='credit']['amount'].sum()
        debits  = df[df['type']=='debit']['amount'].sum()
        by_cat  = df[df['type']=='debit'].groupby('category')['amount'].sum().to_dict()
        return (
            f'Total income: NGN {credits:,.2f}\n'
            f'Total spending: NGN {debits:,.2f}\n'
            f'By category: {by_cat}'
        )


4.4 Opportunity Scout (Background Agent)
# agents/opportunity_scout/agent.py


from agents.base_agent import BaseAgent
from langchain_community.tools import DuckDuckGoSearchRun
from langchain.agents import create_react_agent, AgentExecutor
from langchain_core.prompts import PromptTemplate
from apps.opportunities.models import Opportunity
from apps.users.models import FinancialDNA


SCOUT_PROMPT = PromptTemplate.from_template('''
You are NairaSage's Opportunity Scout. Your job is to find real, current
financial opportunities for this Nigerian user based on their profile.


User profile:
{dna_context}


Search for:
1. Government grants or CBN programs matching their profile
2. New high-yield investment products (T-bills, money market funds)
3. NGX IPOs or newly listed stocks
4. Side income opportunities matching their skills
5. Real estate or agro-investment deals within their budget


For each opportunity found: title, description, source URL, estimated value,
deadline (if any), and a match score 0-100 based on user profile.


Respond in JSON array format.


Tools available: {tools}
Tool names: {tool_names}
{agent_scratchpad}
''')


class OpportunityScoutAgent(BaseAgent):
    agent_name = 'opportunity_scout'


    def run(self, input_data: dict) -> dict:
        search_tool = DuckDuckGoSearchRun()
        agent = create_react_agent(self.llm, [search_tool], SCOUT_PROMPT)
        executor = AgentExecutor(agent=agent, tools=[search_tool],
                                  verbose=False, max_iterations=8)
        result = executor.invoke({
            'dna_context': self._get_dna_context(),
        })
        return self._save_opportunities(result['output'])


    def _save_opportunities(self, raw_output: str) -> dict:
        import json
        try:
            opps = json.loads(raw_output)
            saved = []
            for o in opps:
                obj, created = Opportunity.objects.update_or_create(
                    user_id=self.user_id,
                    title=o.get('title',''),
                    defaults={
                        'opp_type':     o.get('type','investment'),
                        'description':  o.get('description',''),
                        'source_url':   o.get('url',''),
                        'match_score':  o.get('match_score', 50),
                        'urgency_score':o.get('urgency', 0),
                    }
                )
                if created: saved.append(obj.title)
            return {'summary': f'{len(saved)} new opportunities found.', 'saved': saved}
        except Exception as e:
            return {'summary': 'Scout run complete.', 'error': str(e)}

6. Background Tasks — Django-RQ
6.1 Queue Architecture
Django-RQ uses Redis as the message broker. Three queues handle priority separation: high (user-triggered, time-sensitive), default (standard async jobs), and low (batch operations, cron scouts).

    'default': {'USE_REDIS_CACHE': 'default'},  # standard async
    'low':     {'USE_REDIS_CACHE': 'default'},  # scouts, batch, cron
}

                'balance':     t.get('balance',0) / 100,
            }
        )
    # Enqueue AI categorization after sync
    categorize_uncategorized.delay(account.user_id)

from django_rq import get_scheduler
from datetime import datetime, timedelta


def setup_scheduled_tasks():
    scheduler = get_scheduler('low')


    # Clear existing jobs to avoid duplicates on restart
    for job in scheduler.get_jobs():
        job.delete()


    # ── Sync all accounts every 4 hours ─────────────────────
    scheduler.schedule(
        scheduled_time=datetime.utcnow(),
        func='tasks.sync_tasks.sync_all_active_accounts',
        interval=4 * 3600,
        repeat=None,  # infinite
    )


    # ── Opportunity Scout — runs once daily at 06:00 WAT ─────
    scheduler.cron(
        '0 5 * * *',  # 05:00 UTC = 06:00 WAT
        func='tasks.scout_tasks.scout_all_active_users',
        queue_name='low',
    )


    # ── Behavioral check — nightly at 22:00 WAT ──────────────
    scheduler.cron(
        '0 21 * * *',
        func='tasks.scout_tasks.behavioral_check_all_users',
        queue_name='low',
    )


    # ── Market data refresh every 30 mins (trading hours) ────
    scheduler.cron(
        '*/30 9-17 * * 1-5',  # every 30 min, 9am-5pm, Mon-Fri
        func='tasks.market_tasks.refresh_all_portfolios',
        queue_name='default',
    )


    # ── FX rates — every 15 minutes ──────────────────────────
    scheduler.schedule(
        scheduled_time=datetime.utcnow(),
        func='tasks.market_tasks.refresh_fx_rates',
        interval=900,
        repeat=None,

import httpx, os
from typing import Optional


class MonoClient:
    BASE_URL = 'https://api.withmono.com/v2'


    def __init__(self):
        self.api_key = os.getenv('MONO_SECRET_KEY')
        self.headers = {
            'mono-sec-key': self.api_key,
            'Content-Type': 'application/json',
        }


    def exchange_token(self, code: str) -> dict:
        """Exchange Mono Connect auth code for account ID."""
        r = httpx.post(f'{self.BASE_URL}/accounts/auth',
            headers=self.headers, json={'code': code})
        r.raise_for_status()
        return r.json()  # {'id': 'account_id', ...}


    def get_account(self, account_id: str) -> dict:
        r = httpx.get(f'{self.BASE_URL}/accounts/{account_id}',
            headers=self.headers)
        r.raise_for_status()
        return r.json()


    def get_transactions(self, account_id: str, limit: int = 100) -> list:
        r = httpx.get(f'{self.BASE_URL}/accounts/{account_id}/transactions',
            headers=self.headers, params={'limit': limit})
        r.raise_for_status()
        return r.json().get('data', [])


    def get_balance(self, account_id: str) -> Optional[float]:
        data = self.get_account(account_id)
        return data.get('account', {}).get('balance')


    def verify_bvn(self, bvn: str) -> dict:
        r = httpx.post(f'{self.BASE_URL}/v1/identity/bvn/verify',
            headers=self.headers, json={'bvn': bvn})
        r.raise_for_status()
        return r.json()

# integrations/paystack/client.py


import httpx, os, hmac, hashlib


class PaystackClient:
    BASE_URL = 'https://api.paystack.co'


    def __init__(self):
        self.secret = os.getenv('PAYSTACK_SECRET_KEY')
        self.headers = {
            'Authorization': f'Bearer {self.secret}',
            'Content-Type': 'application/json',
        }


    def initialize_transaction(self, email: str, amount_kobo: int,
                                plan_code: str = None) -> dict:
        payload = {'email': email, 'amount': amount_kobo}
        if plan_code: payload['plan'] = plan_code
        r = httpx.post(f'{self.BASE_URL}/transaction/initialize',
            headers=self.headers, json=payload)
        r.raise_for_status()
        return r.json()  # {'authorization_url': '...', 'reference': '...'}


    def verify_transaction(self, reference: str) -> dict:
        r = httpx.get(f'{self.BASE_URL}/transaction/verify/{reference}',
            headers=self.headers)
        r.raise_for_status()
        return r.json()


    def verify_webhook(self, payload: bytes, signature: str) -> bool:
        """Validate Paystack webhook HMAC-SHA512 signature."""
        computed = hmac.new(
            self.secret.encode(), payload, hashlib.sha512
        ).hexdigest()
        return hmac.compare_digest(computed, signature)

import httpx, os
from django.core.cache import cache


class NGXClient:
    """NGX Market Data API — or iTick as MVP fallback."""
    BASE_URL = os.getenv('NGX_API_URL', 'https://marketdataapiv3.ngxgroup.com/api')


    def __init__(self):
        self.api_key = os.getenv('NGX_API_KEY')
        self.headers = {'Authorization': f'Bearer {self.api_key}'}


    def get_price(self, symbol: str) -> float | None:
        cache_key = f'ngx_price_{symbol}'
        cached = cache.get(cache_key)
        if cached: return cached


        try:
            r = httpx.get(f'{self.BASE_URL}/prices/{symbol}',
                headers=self.headers, timeout=10)
            r.raise_for_status()
            price = r.json().get('close')
            cache.set(cache_key, price, timeout=1800)  # 30 min cache
            return price
        except Exception:
            return None


    def get_market_summary(self) -> dict:
        cache_key = 'ngx_market_summary'
        cached = cache.get(cache_key)
        if cached: return cached
        r = httpx.get(f'{self.BASE_URL}/market/summary', headers=self.headers)
        data = r.json()
        cache.set(cache_key, data, timeout=900)
        return data

12. Security Checklist

Item
Implementation
JWT Token Expiry
Access token: 15 min. Refresh token: 7 days. Blacklist on logout (SimpleJWT).
CORS
corsheaders middleware. Whitelist Vercel frontend domain only.
Rate Limiting
django-axes on auth endpoints. DRF throttling: 100/hr anon, 1000/hr authenticated.
Mono Webhooks
Verify HMAC-SHA256 signature on every webhook before processing.
Paystack Webhooks
Verify HMAC-SHA512 signature. Idempotency keys prevent duplicate billing.
SQL Injection
Django ORM used exclusively. No raw SQL. Parameterized queries enforced.
XSS
Next.js escapes JSX by default. Content-Security-Policy header set.
Sensitive Data
Bank credentials never stored (Mono handles OAuth). Balances encrypted at rest.
NDPA Compliance
Explicit consent collected at signup. User can export/delete all data via settings.
Secrets
All keys in environment variables only. Never committed to Git. .env in .gitignore.
HTTPS
Enforced at hosting layer (Vercel + Railway). HSTS header enabled.
Agent Output Safety
All AI responses labeled 'not licensed financial advice'. High-risk actions require user confirmation.
