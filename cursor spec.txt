1. Project Overview & Ground Rules for Cursor
CURSOR INSTRUCTION
You are building NairaSage — an AI-powered personal finance platform for Nigeria. Follow this specification exactly. Do not deviate from the stack, file paths, naming conventions, or patterns described. When in doubt, ask before assuming.


1.1 What You Are Building
NairaSage is a full-stack web application with three layers: (1) a Next.js 14 TypeScript frontend that users interact with, (2) a Django 5 REST API backend that handles all data and business logic, and (3) a multi-agent AI orchestration layer built on LangChain that powers 11 specialist financial AI agents coordinated by a master Coordinator agent.

The system integrates with real Nigerian financial APIs: Mono for open banking, NGX for stock market data, Paystack for payments, and Fixer.io for FX rates. All AI runs through LangChain with Mistral as the default LLM, switchable to OpenAI via a single .env variable.

1.2 Non-Negotiable Rules
CURSOR: FOLLOW THESE EXACTLY
1. Never hardcode API keys — always use environment variables via django-environ (backend) or process.env (frontend). 2. Every Django model must have created_at and updated_at fields. 3. Every API endpoint requires JWT authentication unless explicitly marked Public. 4. All monetary values stored as DecimalField(max_digits=14, decimal_places=2) in NGN. 5. Never store raw bank credentials — Mono handles OAuth. 6. All AI agent outputs must include a disclaimer: 'This is financial information, not licensed advice.' 7. Use Django ORM exclusively — no raw SQL. 8. TypeScript strict mode enabled — no 'any' types.

4.3 LLM Factory (core/llm_factory.py)
CURSOR INSTRUCTION
This is the single most important file in the AI layer. Every agent imports get_llm() from here. Never import LangChain LLM classes directly in agent files.


core/llm_factory.py

import os
from langchain_mistralai import ChatMistralAI
from langchain_openai import ChatOpenAI
from langchain_core.language_models import BaseChatModel


def get_llm(temperature: float = None,
            max_tokens: int = None) -> BaseChatModel:
    """
    Return the configured LLM based on LLM_PROVIDER env var.
    Supports: mistral | openai
    Switch providers by changing LLM_PROVIDER in .env — no code changes.
    """
    provider = os.getenv('LLM_PROVIDER', 'mistral').lower()
    temp     = temperature if temperature is not None \
               else float(os.getenv('LLM_TEMPERATURE', '0.2'))
    tokens   = max_tokens or int(os.getenv('LLM_MAX_TOKENS', '4096'))


    if provider == 'openai':
        return ChatOpenAI(
            model=os.getenv('OPENAI_MODEL', 'gpt-4o'),
            temperature=temp,
            max_tokens=tokens,
            openai_api_key=os.getenv('OPENAI_API_KEY'),
        )


    # Default: Mistral
    return ChatMistralAI(
        model=os.getenv('MISTRAL_MODEL', 'mistral-large-latest'),
        temperature=temp,
        mistral_api_key=os.getenv('MISTRAL_API_KEY'),
    )


def get_routing_llm() -> BaseChatModel:
    """LLM for the coordinator routing step — always temperature=0."""
    return get_llm(temperature=0.0)


def get_creative_llm() -> BaseChatModel:
    """LLM for education/coaching — slightly more creative."""
    return get_llm(temperature=0.5)

6.1 agents/base_agent.py
CURSOR INSTRUCTION
Every specialist agent MUST inherit from BaseAgent. Never call the LLM directly in an agent — always use self.chat() or self.run(). Never import ChatMistralAI or ChatOpenAI directly in agent files.


agents/base_agent.py

from abc import ABC, abstractmethod
from typing import Any, Dict, Optional
import logging
from langchain_core.messages import HumanMessage, SystemMessage
from langchain.memory import ConversationSummaryBufferMemory
from core.llm_factory import get_llm
from apps.users.models import FinancialDNA


logger = logging.getLogger(__name__)


DISCLAIMER = (
    '\n\n---\n'
    'Note: This is financial information provided for educational purposes. '
    'It is not licensed financial advice. Please consult a qualified financial '
    'advisor before making investment or financial decisions.'
)


class BaseAgent(ABC):
    agent_name: str = 'base'
    system_prompt: str = ''
    add_disclaimer: bool = True


    def __init__(self, user_id: int):
        self.user_id = user_id
        self.llm     = get_llm()
        try:
            self.dna = FinancialDNA.objects.select_related('user').get(
                user_id=user_id
            )
        except FinancialDNA.DoesNotExist:
            # Create empty DNA if onboarding not complete
            from apps.users.models import User
            user = User.objects.get(id=user_id)
            self.dna = FinancialDNA.objects.create(user=user)
        self.memory = self._load_memory()


    def _load_memory(self) -> ConversationSummaryBufferMemory:
        mem = ConversationSummaryBufferMemory(
            llm=self.llm,
            max_token_limit=2000,
            return_messages=True,
        )
        saved = self.dna.agent_memory.get(self.agent_name, {})
        if saved.get('summary'):
            mem.moving_summary_buffer = saved['summary']
        return mem


    def _save_memory(self):
        self.dna.agent_memory[self.agent_name] = {
            'summary': self.memory.moving_summary_buffer,
            'updated': str(import_datetime()),
        }
        self.dna.save(update_fields=['agent_memory', 'updated_at'])


    def _get_dna_context(self) -> str:
        return (
            f'USER PROFILE:\n'
            f'- Monthly income: NGN {self.dna.monthly_income}\n'
            f'- Risk tolerance: {self.dna.risk_tolerance}\n'
            f'- Primary goal: {self.dna.primary_goal}\n'
            f'- Life stage: {self.dna.life_stage}\n'
            f'- Financial literacy: {self.dna.financial_literacy_score}/100\n'
            f'- Occupation: {self.dna.occupation}\n'
        )


    def chat(self, user_input: str) -> str:
        """Standard chat with persistent memory. Returns response string."""
        history  = self.memory.load_memory_variables({}).get('history', [])
        messages = [
            SystemMessage(content=(
                self.system_prompt + '\n\n' + self._get_dna_context()
            )),
            *history,
            HumanMessage(content=user_input),
        ]
        try:
            response = self.llm.invoke(messages)
            content  = response.content
            if self.add_disclaimer:
                content += DISCLAIMER
            self.memory.save_context(
                {'input': user_input}, {'output': content}
            )
            self._save_memory()
            return content
        except Exception as e:
            logger.error(f'{self.agent_name} error: {e}')
            return f'I encountered an error processing your request. Please try again.'


    @abstractmethod
    def run(self, input_data: Dict[str, Any]) -> Dict[str, Any]:
        """
        Execute the agent's primary task.
        Must return: {'summary': str, 'data': dict, 'confidence': float}
        """
        pass


def import_datetime():
    from datetime import datetime
    return datetime.utcnow().isoformat()

import json
import logging
from typing import List, Dict, Any
from langchain_core.messages import SystemMessage, HumanMessage
from core.llm_factory import get_routing_llm
from apps.users.models import FinancialDNA, SkillOutcome, SkillGapReport
import time


logger = logging.getLogger(__name__)


# ── Agent registry ────────────────────────────────────────
# Import lazily inside dispatch() to avoid circular imports
AGENT_REGISTRY = {
    'accountant':         'agents.accountant.agent.AccountantAgent',
    'economist':          'agents.economist.agent.EconomistAgent',
    'investment_advisor': 'agents.investment_advisor.agent.InvestmentAdvisorAgent',
    'opportunity_scout':  'agents.opportunity_scout.agent.OpportunityScoutAgent',
    'tax_advisor':        'agents.tax_advisor.agent.TaxAdvisorAgent',
    'behavioral_coach':   'agents.behavioral_coach.agent.BehavioralCoachAgent',
    'debt_manager':       'agents.debt_manager.agent.DebtManagerAgent',
    'savings_optimizer':  'agents.savings_optimizer.agent.SavingsOptimizerAgent',
    'risk_assessor':      'agents.risk_assessor.agent.RiskAssessorAgent',
    'life_planner':       'agents.life_planner.agent.LifePlannerAgent',
    'education':          'agents.education.agent.EducationAgent',
}


ROUTING_SYSTEM_PROMPT = '''
You are the NairaSage coordinator. Analyze the user's message and return
ONLY a JSON object with two fields:
  agents: array of agent names to call (1-3 agents max)
  intent_bucket: a short snake_case label for this query type


Available agents:
  accountant         - spending, transactions, budgeting, cash flow
  economist          - inflation, macro, Naira, CBN policy, FX
  investment_advisor - stocks, portfolio, NGX, mutual funds, T-bills
  opportunity_scout  - grants, deals, side income, IPOs
  tax_advisor        - PAYE, self-assessment, FIRS, deductions
  behavioral_coach   - impulse spending, habits, emotional decisions
  debt_manager       - loans, repayment, interest rates
  savings_optimizer  - goals, high-yield options, auto-save
  risk_assessor      - portfolio risk, diversification, stress testing
  life_planner       - retirement, house, long-term goals
  education          - learning financial concepts, explanations


Examples:
  'how much did I spend?' -> {"agents":["accountant"],"intent_bucket":"spending_query"}
  'should I buy MTNN?'    -> {"agents":["investment_advisor","risk_assessor"],"intent_bucket":"stock_recommendation"}
  'plan my retirement'    -> {"agents":["life_planner","savings_optimizer"],"intent_bucket":"retirement_planning"}
  'I keep overspending'   -> {"agents":["behavioral_coach","accountant"],"intent_bucket":"behavioral_flag"}
  'what is a P/E ratio?'  -> {"agents":["education"],"intent_bucket":"concept_explanation"}


Return ONLY valid JSON. No explanation.
'''


SYNTHESIS_PROMPT = '''
You are NairaSage, a personal financial co-pilot for Nigerians.
Synthesize the specialist agent reports below into ONE clear, actionable response.
Rules:
- Write as if speaking directly to the user
- Use Nigerian context (NGN, NGX, T-bills, CBN etc)
- Be specific with numbers — never be vague
- Calibrate complexity to user literacy score: {literacy_score}/100
- If literacy < 40: use very simple language and analogies
- If literacy 40-70: explain terms briefly
- If literacy > 70: use proper financial terminology
- End with one clear next action the user should take
'''


class CoordinatorAgent:
    def __init__(self, user_id: int):
        self.user_id = user_id
        self.llm     = get_routing_llm()
        self.dna     = FinancialDNA.objects.get(user_id=user_id)


    def route(self, message: str) -> Dict[str, Any]:
        """Determine which agents to call. Returns routing decision."""
        # Check routing_weights for learned preferences
        response = self.llm.invoke([
            SystemMessage(content=ROUTING_PROMPT),
            HumanMessage(content=message),
        ])
        try:
            decision = json.loads(response.content)
            # Apply learned routing weights to re-rank agents
            decision = self._apply_routing_weights(decision)
            return decision
        except json.JSONDecodeError:
            return {'agents': ['accountant'], 'intent_bucket': 'fallback'}


    def _apply_routing_weights(self, decision: dict) -> dict:
        """Boost or suppress agents based on learned weights."""
        weights = self.dna.routing_weights
        bucket  = decision.get('intent_bucket', '')
        if bucket in weights:
            # Add any consistently high-performing agents for this bucket
            for agent, weight in weights[bucket].items():
                if weight > 0.8 and agent not in decision['agents']:
                    decision['agents'].append(agent)
        return decision


    def _load_agent(self, agent_path: str, user_id: int):
        """Dynamically import and instantiate an agent class."""
        module_path, class_name = agent_path.rsplit('.', 1)
        import importlib
        module = importlib.import_module(module_path)
        return getattr(module, class_name)(user_id=user_id)


    def dispatch(self, message: str) -> Dict[str, Any]:
        """Main entry point. Route, call agents, synthesize."""
        start    = time.time()
        routing  = self.route(message)
        agents   = routing.get('agents', ['accountant'])
        bucket   = routing.get('intent_bucket', 'unknown')


        # Run agents
        results  = {}
        for name in agents:
            path = AGENT_REGISTRY.get(name)
            if not path: continue
            try:
                agent = self._load_agent(path, self.user_id)
                results[name] = agent.run({'query': message})
            except Exception as e:
                logger.error(f'Agent {name} failed: {e}')
                results[name] = {'summary': f'[{name} unavailable]', 'data': {}}


        # Synthesize
        response = self._synthesize(message, results)


        # Log outcome for self-learning
        elapsed = int((time.time() - start) * 1000)
        SkillOutcome.objects.create(
            user_id=self.user_id,
            intent_bucket=bucket,
            agents_called=agents,
            user_query=message[:500],
            response_time_ms=elapsed,
        )


        # Check for skill gaps
        if not results or all('[unavailable]' in r.get('summary','') for r in results.values()):
            self._flag_skill_gap(message, bucket)


        return {
            'response':    response,
            'agents_used': agents,
            'intent':      bucket,
        }


    def _synthesize(self, query: str, results: dict) -> str:
        context = '\n\n'.join([
            f'[{name} report]: {r.get("summary","")}' 
            for name, r in results.items()
        ])
        literacy = self.dna.financial_literacy_score
        response = self.llm.invoke([
            SystemMessage(content=SYNTHESIS_PROMPT.format(
                literacy_score=literacy
            )),
            HumanMessage(content=(
                f'User asked: {query}\n\nAgent reports:\n{context}'
            )),
        ])
        return response.content


    def _flag_skill_gap(self, query: str, bucket: str):
        """Flag this query type for David's review if unhandleable."""
        existing = SkillGapReport.objects.filter(
            query_pattern__icontains=bucket, status='pending'
        ).first()
        if existing:
            existing.frequency += 1
            existing.save(update_fields=['frequency'])
        else:
            # Ask LLM to write a spec for the missing skill
            spec_response = self.llm.invoke([
                SystemMessage(content='Write a brief technical spec for a new NairaSage agent skill that would handle this type of user query. Include: what data it needs, what APIs to call, what output to produce. Keep it under 200 words.'),
                HumanMessage(content=f'Query: {query}\nBucket: {bucket}'),
            ])
            SkillGapReport.objects.create(
                query_pattern=query[:300],
                proposed_spec=spec_response.content,
            )


    def record_feedback(self, outcome_id: int, was_helpful: bool):
        """Record user thumbs up/down — triggers learning update."""
        SkillOutcome.objects.filter(id=outcome_id).update(
            was_helpful=was_helpful
        )
        # Queue the learning update job
        from tasks.scout_tasks import update_routing_weights
        update_routing_weights.delay(self.user_id)

# agents/skills/financial_math.py


import numpy as np
from decimal import Decimal
from typing import List, Dict


def compound_growth(principal: float, rate: float,
                    years: int, contributions: float = 0) -> Dict:
    """Calculate compound growth with optional monthly contributions."""
    monthly_rate = rate / 12 / 100
    months = years * 12
    balance = principal
    history = []
    for m in range(months):
        balance = balance * (1 + monthly_rate) + contributions
        if m % 12 == 11:
            history.append(round(balance, 2))
    return {'final': round(balance, 2), 'history': history}


def real_return(nominal_return_pct: float,
               inflation_pct: float) -> float:
    """Fisher equation: real return adjusted for inflation."""
    return ((1 + nominal_return_pct/100) /
            (1 + inflation_pct/100) - 1) * 100


def debt_payoff_schedule(principal: float, annual_rate: float,
                          monthly_payment: float) -> Dict:
    """Calculate months to payoff and total interest paid."""
    monthly_rate = annual_rate / 12 / 100
    balance = principal
    months = 0
    total_interest = 0
    while balance > 0 and months < 600:
        interest = balance * monthly_rate
        principal_paid = monthly_payment - interest
        balance -= principal_paid
        total_interest += interest
        months += 1
    return {
        'months': months,
        'total_interest': round(total_interest, 2),
        'total_paid': round(principal + total_interest, 2),
    }


def sharpe_ratio(returns: List[float], risk_free: float = 0.18) -> float:
    """Sharpe ratio. risk_free = Nigeria T-bill rate (default 18%)."""
    if not returns or len(returns) < 2:
        return 0.0
    arr = np.array(returns)
    excess = arr - (risk_free / 12)
    return float(np.mean(excess) / (np.std(excess) + 1e-9))


def portfolio_var(returns: List[float], confidence: float = 0.95) -> float:
    """Value at Risk at given confidence level."""
    if not returns:
        return 0.0
    return float(np.percentile(np.array(returns), (1 - confidence) * 100))


def paye_nigeria(annual_gross: float) -> Dict:
    """Calculate Nigerian PAYE tax on annual gross income (2024 rates)."""
    # PITA tax bands
    bands = [
        (300_000,   7),
        (300_000,  11),
        (500_000,  15),
        (500_000,  19),
        (1_600_000,21),
        (float('inf'), 24),
    ]
    # CRA: higher of NGN 200k or 1% of gross income + 20% of gross
    cra = max(200_000, 0.01 * annual_gross) + 0.20 * annual_gross
    taxable = max(0, annual_gross - cra)
    tax = 0
    remaining = taxable
    for band_size, rate in bands:
        if remaining <= 0: break
        chunk = min(remaining, band_size)
        tax  += chunk * rate / 100
        remaining -= chunk
    return {
        'gross': annual_gross,
        'cra': round(cra, 2),
        'taxable': round(taxable, 2),
        'tax': round(tax, 2),
        'effective_rate': round(tax / annual_gross * 100, 2) if annual_gross else 0,
        'monthly_paye': round(tax / 12, 2),
    }

from rest_framework import status
from agents.coordinator import CoordinatorAgent, AGENT_REGISTRY

class CoordinatorChatView(APIView):
    def post(self, request):
        message = request.data.get('message', '').strip()
        if not message:
            return Response(
                {'error': 'message is required'},
                status=status.HTTP_400_BAD_REQUEST
            )
        coordinator = CoordinatorAgent(user_id=request.user.id)
        result = coordinator.dispatch(message)
        return Response(result)




class AgentDirectView(APIView):
    def post(self, request, agent: str):
        message = request.data.get('message', '').strip()
        path = AGENT_REGISTRY.get(agent)
        if not path:
            return Response(
                {'error': f'Unknown agent: {agent}'},
                status=status.HTTP_404_NOT_FOUND
            )
        import importlib
        mod_path, cls = path.rsplit('.', 1)
        module   = importlib.import_module(mod_path)
        AgentCls = getattr(module, cls)
        agent_obj = AgentCls(user_id=request.user.id)
        result = agent_obj.run({'query': message})
        return Response(result)




class FeedbackView(APIView):
    def post(self, request, outcome_id: int):
        helpful = request.data.get('helpful', None)
        if helpful is None:
            return Response({'error': 'helpful field required'})
        coordinator = CoordinatorAgent(user_id=request.user.id)
        coordinator.record_feedback(outcome_id, bool(helpful))
        return Response({'status': 'recorded'})




class AgentHistoryView(APIView):
    def get(self, request, agent: str):
        from apps.users.models import FinancialDNA
        dna    = FinancialDNA.objects.get(user_id=request.user.id)
        memory = dna.agent_memory.get(agent, {})
        return Response({'agent': agent, 'memory': memory})




class ClearHistoryView(APIView):
    def delete(self, request, agent: str):
        from apps.users.models import FinancialDNA
        dna = FinancialDNA.objects.get(user_id=request.user.id)
        dna.agent_memory.pop(agent, None)
        dna.save(update_fields=['agent_memory'])
        return Response({'status': 'cleared'})

@django_rq.job('default')
def sync_account_transactions(account_id: int):
    account = BankAccount.objects.get(id=account_id)
    client  = MonoClient()
    try:
        txns = client.get_transactions(
            account.mono_account_id, limit=100
        )
        created = 0
        for t in txns:
            _, was_created = Transaction.objects.update_or_create(
                mono_id=t['_id'],
                defaults={
                    'account':     account,
                    'amount':      abs(t['amount']) / 100,
                    'type':        'credit' if t['amount'] > 0 else 'debit',
                    'description': t.get('narration', ''),
                    'date':        t['date'][:10],
                    'balance':     t.get('balance', 0) / 100,
                }
            )
            if was_created: created += 1
        # Queue AI categorization for new transactions
        if created > 0:
            categorize_new_transactions.delay(account.user_id)
        logger.info(f'Synced {created} new transactions for account {account_id}')
    except Exception as e:
        logger.error(f'Sync failed for account {account_id}: {e}')

@django_rq.job('low')
def update_routing_weights(user_id: int):
    """
    Self-learning: analyze SkillOutcome records to improve
    routing weights in the user's Financial DNA.
    Run after each feedback event.
    """
    from collections import defaultdict
    outcomes = SkillOutcome.objects.filter(
        user_id=user_id,
        was_helpful__isnull=False
    ).values('intent_bucket', 'agents_called', 'was_helpful')


    bucket_stats = defaultdict(lambda: defaultdict(lambda: {'pos':0,'neg':0}))
    for o in outcomes:
        for agent in o['agents_called']:
            if o['was_helpful']:
                bucket_stats[o['intent_bucket']][agent]['pos'] += 1
            else:
                bucket_stats[o['intent_bucket']][agent]['neg'] += 1


    new_weights = {}
    for bucket, agents in bucket_stats.items():
        new_weights[bucket] = {}
        for agent, counts in agents.items():
            total = counts['pos'] + counts['neg']
            if total >= 3:  # minimum samples
                new_weights[bucket][agent] = round(
                    counts['pos'] / total, 3
                )


    dna = FinancialDNA.objects.get(user_id=user_id)
    dna.routing_weights = new_weights
    dna.save(update_fields=['routing_weights', 'updated_at'])
    logger.info(f'Updated routing weights for user {user_id}')

from django_rq import get_scheduler
from datetime import datetime


def setup_scheduled_tasks():
    scheduler = get_scheduler('low')
    # Clear existing to avoid duplicates on restart
    for job in scheduler.get_jobs(): job.delete()


    # Sync bank accounts every 4 hours
    scheduler.schedule(
        scheduled_time=datetime.utcnow(),
        func='tasks.sync_tasks.sync_all_active_accounts',
        interval=4 * 3600, repeat=None,
    )
    # Opportunity scout daily at 06:00 WAT (05:00 UTC)
    scheduler.cron('0 5 * * *',
        func='tasks.scout_tasks.scout_all_active_users',
        queue_name='low')
    # FX rates every 15 min
    scheduler.schedule(
        scheduled_time=datetime.utcnow(),
        func='tasks.market_tasks.refresh_fx_rates',
        interval=900, repeat=None,
    )
    # Market prices every 30 min on trading days
    scheduler.cron('*/30 9-17 * * 1-5',
        func='tasks.market_tasks.refresh_all_portfolios',
        queue_name='default')
